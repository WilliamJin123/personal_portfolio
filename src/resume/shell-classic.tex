% ran the Winter 2025 cycle), rebuilt on the generator. It coexists with

% Faithful to the old design (airier leading, \LARGE ruled section titles,
% dark-blue links, label-column skills with bold space-separated tags,
% italic context line under projects) EXCEPT the ATS decisions, which stay:
% letterpaper, no FontAwesome icons (private-use glyphs break extraction),
% TeX Gyre Heros with common ligatures off, text link labels instead of icons.

% Text alignment

\setlength{\tabcolsep}{0in}

% Section formatting — the \LARGE ruled titles are the design's signature.
% -10pt pre-space: titles sit close to the previous section's last line; the
% reclaimed points buy generous spacing INSIDE Experience/Projects instead
% (entry pre-gaps, heading→bullet gaps, itemsep — William's redistribution).
\titleformat{\section}
  {\vspace{-10pt}\raggedright\LARGE\mdseries}{}{0em}{}

% Custom commands
\newcommand{\resumeItem}[2]{
  \item{
    \large \textbf{#1}{\hspace{0.5mm}\large #2 \vspace{-0.5mm}}
  }
}

\newcommand{\resumePOR}[3]{
\vspace{0.5mm}\item[]
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{#1}\hspace{0.3mm}#2 & \textit{\small{#3}}
    \end{tabular*}
    \vspace{-2mm}
}

% Classic heading line: \large bold title, body-size role, footnotesize italic
% date — the old size hierarchy. #2 is unused; the generator folds the location
% into the right-hand cell ("City, PR | Mon YYYY - Mon YYYY") via #4.
\newcommand{\resumeSubheading}[4]{
\vspace{1mm}\item[]
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        \textbf{\large #1} | #3 & \textit{\footnotesize{#4}} \\
    \end{tabular*}
    \vspace{-1.9mm}
}

% Project heading. #1 = title, formatted by the generator (bold \large name plus
% small trailing link labels — the ATS-safe replacement for the old external-link
% icon); #2 = date; #3 = the italic context row (award / grant / event /
% "Personal Project"), which carries its own leading \\ so a context-less
% project stays a single clean row.
\newcommand{\resumeProject}[3]{
\vspace{1mm}\item[]
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        {#1} & \textit{\footnotesize{#2}} #3
    \end{tabular*}
    \vspace{-1.9mm}
}

\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}\vspace{-4pt}}

\renewcommand{\labelitemi}{$\vcenter{\hbox{\tiny$\bullet$}}$}

% the tighter section-boundary spacing above.
\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=5mm,labelsep=3mm,itemsep=0.4mm]}

% halved here — still a visible breath before the next ruled title.

\newcommand{\resumeHeadingSkillEnd}{\end{itemize}\vspace{-3.5mm}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-3.9mm}}

% Document font: TeX Gyre Heros — the OpenType successor of the NimbusSanL the
% W25 résumé was set in, so the look is identical, and \textbf keeps working
% under tectonic (the Type1 phv family silently loses bold once fontspec is
% active). Ligatures={TeX, NoCommon} keeps -- mappings but disables ff/fi/fl
% ligatures, whose U+FB00-class glyphs break ATS keyword matching.
\newfontfamily\resumemain{texgyreheros}[
  Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic,
  Ligatures={TeX, NoCommon}]
