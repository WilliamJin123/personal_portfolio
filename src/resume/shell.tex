\newcommand{\resumePOR}[3]{
\vspace{0.5mm}\item[]
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}

    \end{tabular*}
    \vspace{-2mm}
}

% One heading line, one size: company bold, role regular, date italic — separated
% by weight/slant, not size (per review). #2 (location) is intentionally unused.
\newcommand{\resumeSubheading}[4]{
\vspace{1.5mm}\item[]
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        {\headsize\textbf{#1} | #3} & {\headsize\textit{#4}} \\
    \end{tabular*}
    \vspace{-2.2mm}
}

% Project heading. #1 = title (bold repo-link) optionally "| event" inline, all at
% \headsize; #2 = date (italic, same size); #3 = optional badge row, which carries
% its own leading \\ so a project with no awards/grants stays a single clean row.
\newcommand{\resumeProject}[3]{
\vspace{1.5mm}\item[]
    \begin{tabular*}{0.98\textwidth}[t]{l@{\extracolsep{\fill}}r}
        {\headsize #1} & {\headsize\textit{#2}} #3
    \end{tabular*}
    \vspace{-2.2mm}
}

% Document font: TeX Gyre Heros (a Helvetica twin) loaded as OpenType via fontspec.
% Keeps the Helvetica look AND working \textbf under tectonic — the Type1 phv/qhv
% Helvetica families silently lose their bold series once fontspec is active (which
% fontawesome pulls in under XeTeX). fontspec is already loaded by fontawesome above.
\newfontfamily\resumemain{texgyreheros}[
  Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\newcommand{\headerfontxi}{\resumemain}

% \award (trophy) and \grant ($) tag wins and funding on a project's second row.
% One ink colour now — the icon, not colour, marks them (per review); kept bold.
% \resumemain (sans body family), not \normalfont — \normalfont snaps back to the
% serif default and the badges would clash with the sans body. Kept subtle (per
% review): regular weight, no bold, and the trophy pictograph a notch smaller so
% the solid glyph stops shouting over its label.
\newcommand{\award}[1]{{\resumemain{\footnotesize\faTrophy}~#1}\hspace{7pt}}
\newcommand{\grant}[1]{{\resumemain\faUsd #1}\hspace{7pt}}
% \weblink: a small underlined link for an extra artifact (paper, live demo) that is
% neither the title's repo nor the event's Devpost. Underlined, no icon (per review).
\newcommand{\weblink}[2]{{\resumemain\footnotesize\reslink{#1}{#2}}\hspace{6pt}}

% \wjicon: the site favicon (the "WJ" browser-tab monogram) redrawn in TikZ so the
% portfolio link carries the same mark as the tab. Palette matches public/favicon.svg
% but INVERTED — a solid dark ink square with the W in paper and the J in amber — so
% it reads as a dark badge in the same set as the LinkedIn/GitHub squares rather than
% a faint light box, while keeping the amber-J identity. \ttfamily is Latin Modern
% Mono (bundled) — robust under XeTeX.
%
% \wjbadge draws it once at a fixed 12pt square with locked internal proportions;
% \wjicon then runs that through the SAME wrapper \socialicon uses for the brand
% glyphs — \raisebox{-0.05em}{\resizebox{!}{1em}{...}} — so it normalizes to the
% exact size and baseline of the LinkedIn/GitHub badges automatically.
\definecolor{wjpaper}{HTML}{ECEEF1}
\definecolor{wjink}{HTML}{16181C}
\definecolor{wjamber}{HTML}{BD741B}
\newcommand{\wjbadge}{%
  \begin{tikzpicture}
    \node[rounded corners=2.4pt, fill=wjink, minimum size=12pt, inner sep=0pt]
      {\ttfamily\bfseries\fontsize{7pt}{7pt}\selectfont
       \textcolor{wjpaper}{WJ}};
  \end{tikzpicture}}
\newcommand{\wjicon}{\raisebox{-0.05em}{\resizebox{!}{1em}{\wjbadge}}}
